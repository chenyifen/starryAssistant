## 🎯 测试目标

### 核心功能测试
1. **唤醒词检测稳定性**
   - 连续唤醒成功率
   - 误唤醒率
   - 唤醒延迟

2. **ASR识别准确性**
   - 识别准确率
   - 识别延迟
   - 部分识别结果连续性

3. **技能匹配和执行**
   - 匹配准确率
   - 执行成功率
   - 响应时间

### 稳定性测试
1. **长时间运行**
   - 24小时连续运行
   - 内存泄漏检测
   - CPU使用率监控

2. **高频操作**
   - 快速连续唤醒
   - 并发操作处理
   - 资源竞争

3. **边界条件**
   - 异常音频输入
   - 网络异常
   - 资源不足

```python
# test_framework/audio_player.py
import pyaudio
import wave
import threading
import time
from typing import Optional

class AudioPlayer:
    """音频播放器"""
    
    def __init__(self, device_index: Optional[int] = None):
        self.pyaudio = pyaudio.PyAudio()
        self.device_index = device_index
        self.is_playing = False
    
    def play_wake_word(self, audio_file: str, volume: float = 1.0):
        """播放唤醒词"""
        threading.Thread(
            target=self._play_audio,
            args=(audio_file, volume),
            daemon=True
        ).start()
    
    def play_command(self, audio_file: str, delay_ms: int = 0):
        """播放命令音频（带延迟）"""
        if delay_ms > 0:
            time.sleep(delay_ms / 1000.0)
        self.play_wake_word(audio_file)
    
    def _play_audio(self, audio_file: str, volume: float):
        """内部播放实现"""
        wf = wave.open(audio_file, 'rb')
        stream = self.pyaudio.open(
            format=self.pyaudio.get_format_from_width(wf.getsampwidth()),
            channels=wf.getnchannels(),
            rate=wf.getframerate(),
            output=True,
            output_device_index=self.device_index
        )
        
        chunk = 1024
        data = wf.readframes(chunk)
        
        while data:
            # 应用音量控制
            if volume != 1.0:
                import numpy as np
                audio_data = np.frombuffer(data, dtype=np.int16)
                audio_data = (audio_data * volume).astype(np.int16)
                data = audio_data.tobytes()
            
            stream.write(data)
            data = wf.readframes(chunk)
        
        stream.stop_stream()
        stream.close()
        wf.close()
```

            '-t', 'audio/wav'
        ])

### 3. 日志收集和监控

#### 日志收集器

```python
# test_framework/log_collector.py
import subprocess
import threading
import queue
import re
from datetime import datetime
from typing import Dict, List

class LogCollector:
    """日志收集器"""
    
    def __init__(self, device_serial: str):
        self.device_serial = device_serial
        self.log_queue = queue.Queue()
        self.collecting = False
        self.thread = None
    
    def start_collecting(self, filters: List[str] = None):
        """开始收集日志"""
        self.collecting = True
        self.thread = threading.Thread(
            target=self._collect_logs,
            args=(filters,),
            daemon=True
        )
        self.thread.start()
    
    def stop_collecting(self):
        """停止收集日志"""
        self.collecting = False
        if self.thread:
            self.thread.join()
    
    def _collect_logs(self, filters: List[str]):
        """内部日志收集"""
        cmd = ['adb', '-s', self.device_serial, 'logcat']
        if filters:
            cmd.extend(filters)
        
        process = subprocess.Popen(
            cmd,
            stdout=subprocess.PIPE,
            stderr=subprocess.PIPE,
            text=True,
            bufsize=1
        )
        
        while self.collecting:
            line = process.stdout.readline()
            if line:
                self.log_queue.put({
                    'timestamp': datetime.now().isoformat(),
                    'line': line.strip()
                })
    
    def extract_metrics(self) -> Dict:
        """提取性能指标"""
        metrics = {
            'wake_word_detections': 0,
            'asr_results': 0,
            'skill_executions': 0,
            'errors': 0,
            'memory_usage': [],
            'cpu_usage': []
        }
        
        while not self.log_queue.empty():
            log_entry = self.log_queue.get()
            line = log_entry['line']
            
            # 统计唤醒词检测
            if 'WAKE WORD DETECTED' in line:
                metrics['wake_word_detections'] += 1
            
            # 统计ASR结果
            if 'ASR Final Result' in line:
                metrics['asr_results'] += 1
            
            # 统计技能执行
            if 'Skill executed' in line:
                metrics['skill_executions'] += 1
            
            # 统计错误
            if 'ERROR' in line or 'Exception' in line:
                metrics['errors'] += 1
            
            # 提取内存使用
            mem_match = re.search(r'memory: (\d+)MB', line)
            if mem_match:
                metrics['memory_usage'].append({
                    'timestamp': log_entry['timestamp'],
                    'value': int(mem_match.group(1))
                })
        
        return metrics
```

# test_framework/performance_monitor.py
import subprocess
import time
import json
from typing import Dict, List
from dataclasses import dataclass, asdict

@dataclass
class PerformanceMetrics:
    """性能指标"""
    timestamp: str
    cpu_usage: float
    memory_mb: float
    battery_level: int
    temperature: float
    wake_word_detections: int
    asr_latency_ms: float

class PerformanceMonitor:
    """性能监控器"""
    
    def __init__(self, device_serial: str):
        self.device_serial = device_serial
        self.metrics_history: List[PerformanceMetrics] = []
    
    def collect_metrics(self) -> PerformanceMetrics:
        """收集当前性能指标"""
        # CPU使用率
        cpu_usage = self._get_cpu_usage()
        
        # 内存使用
        memory_mb = self._get_memory_usage()
        
        # 电池电量
        battery_level = self._get_battery_level()
        
        # 温度
        temperature = self._get_temperature()
        
        return PerformanceMetrics(
            timestamp=time.strftime('%Y-%m-%d %H:%M:%S'),
            cpu_usage=cpu_usage,
            memory_mb=memory_mb,
            battery_level=battery_level,
            temperature=temperature,
            wake_word_detections=0,
            asr_latency_ms=0.0
        )
    
    def _get_cpu_usage(self) -> float:
        """获取CPU使用率"""
        result = subprocess.run([
            'adb', '-s', self.device_serial,
            'shell', 'top', '-n', '1', '-d', '1'
        ], capture_output=True, text=True)
        
        # 解析top输出
        # 返回应用CPU使用率
        return 0.0  # 实现解析逻辑
    
    def _get_memory_usage(self) -> float:
        """获取内存使用（MB）"""
        result = subprocess.run([
            'adb', '-s', self.device_serial,
            'shell', 'dumpsys', 'meminfo', 'com.ai.voice'
        ], capture_output=True, text=True)
        
        # 解析meminfo输出
        # 返回PSS内存（MB）
        return 0.0  # 实现解析逻辑
    
    def _get_battery_level(self) -> int:
        """获取电池电量"""
        result = subprocess.run([
            'adb', '-s', self.device_serial,
            'shell', 'dumpsys', 'battery'
        ], capture_output=True, text=True)
        
        # 解析battery输出
        level_match = re.search(r'level: (\d+)', result.stdout)
        return int(level_match.group(1)) if level_match else 0
    
    def _get_temperature(self) -> float:
        """获取设备温度"""
        result = subprocess.run([
            'adb', '-s', self.device_serial,
            'shell', 'cat', '/sys/class/thermal/thermal_zone0/temp'
        ], capture_output=True, text=True)
        
        # 温度通常是毫摄氏度
        return float(result.stdout.strip()) / 1000.0 if result.stdout.strip() else 0.0
```

async def test_wake_word_stability(
    runner: StressTestRunner,
    duration_minutes: int = 60,
    interval_seconds: int = 5
):
    """测试唤醒词稳定性"""
    total_attempts = (duration_minutes * 60) // interval_seconds
    successful_detections = 0
    
    for i in range(total_attempts):
        # 播放唤醒词
        runner.audio_source.play_wake_word(
            runner.audio_source.wake_word_file
        )
        
        # 等待检测（最多3秒）
        await asyncio.sleep(3)
        
        # 检查是否检测到
        if check_wake_word_detected(runner.devices[0]):
            successful_detections += 1
        
        # 等待间隔
        await asyncio.sleep(interval_seconds)
    
    success_rate = successful_detections / total_attempts
    return {
        'total_attempts': total_attempts,
        'successful_detections': successful_detections,
        'success_rate': success_rate
    }

async def test_asr_accuracy(
    runner: StressTestRunner,
    test_cases: List[Dict[str, str]]
):
    """测试ASR识别准确性"""
    results = []
    
    for test_case in test_cases:
        # 触发唤醒
        runner.audio_source.play_wake_word(
            runner.audio_source.wake_word_file
        )
        await asyncio.sleep(1)
        
        # 播放命令音频
        runner.audio_source.play_command(test_case['audio_file'])
        
        # 等待识别完成
        await asyncio.sleep(5)
        
        # 获取识别结果
        asr_result = get_asr_result(runner.devices[0])
        
        # 比较结果
        is_correct = asr_result.lower() == test_case['expected_text'].lower()
        
        results.append({
            'test_case': test_case['name'],
            'expected': test_case['expected_text'],
            'actual': asr_result,
            'correct': is_correct
        })
    
    accuracy = sum(1 for r in results if r['correct']) / len(results)
    return {
        'total_cases': len(results),
        'correct_cases': sum(1 for r in results if r['correct']),
        'accuracy': accuracy,
        'details': results
    }
```

async def test_long_running(
    runner: StressTestRunner,
    duration_hours: int = 24,
    check_interval_minutes: int = 30
):
    """长时间运行测试"""
    start_time = time.time()
    end_time = start_time + (duration_hours * 3600)
    
    check_count = (duration_hours * 60) // check_interval_minutes
    metrics_history = []
    
    # 启动性能监控
    for device in runner.devices:
        monitor = PerformanceMonitor(device.adb_serial)
        monitor.start_monitoring()
    
    while time.time() < end_time:
        # 定期检查
        for i in range(check_count):
            await asyncio.sleep(check_interval_minutes * 60)
            
            # 收集性能指标
            for device in runner.devices:
                monitor = PerformanceMonitor(device.adb_serial)
                metrics = monitor.collect_metrics()
                metrics_history.append(metrics)
            
            # 执行一次唤醒测试
            runner.audio_source.play_wake_word(
                runner.audio_source.wake_word_file
            )
            await asyncio.sleep(3)
    
    # 分析结果

async def test_high_frequency_operations(
    runner: StressTestRunner,
    operations_per_second: int = 10,
    duration_seconds: int = 60
):
    """高频操作测试"""
    total_operations = operations_per_second * duration_seconds
    interval_ms = 1000 // operations_per_second
    
    successful_operations = 0
    failed_operations = 0
    
    for i in range(total_operations):
        try:
            # 快速连续唤醒
            runner.audio_source.play_wake_word(
                runner.audio_source.wake_word_file
            )
            
            await asyncio.sleep(interval_ms / 1000.0)
            
            if check_wake_word_detected(runner.devices[0]):
                successful_operations += 1
            else:
                failed_operations += 1
        except Exception as e:
            failed_operations += 1
            print(f"Operation {i} failed: {e}")
    
    return {
        'total_operations': total_operations,
        'successful': successful_operations,
        'failed': failed_operations,
        'success_rate': successful_operations / total_operations
