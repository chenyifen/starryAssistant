### 2. 激活状态加密存储

#### 2.1 使用Android Keystore加密

```kotlin
// SecureStorage.kt
import android.content.Context
import android.security.keystore.KeyGenParameterSpec
import android.security.keystore.KeyProperties
import javax.crypto.Cipher
import javax.crypto.KeyGenerator
import javax.crypto.SecretKey
import javax.crypto.spec.GCMParameterSpec
import java.security.KeyStore
import android.util.Base64

class SecureStorage(private val context: Context) {
    private val keyStore = KeyStore.getInstance("AndroidKeyStore").apply { load(null) }
    private val keyAlias = "activation_key"
    
    private fun getSecretKey(): SecretKey {
        if (!keyStore.containsAlias(keyAlias)) {
            val keyGenerator = KeyGenerator.getInstance(
                KeyProperties.KEY_ALGORITHM_AES, 
                "AndroidKeyStore"
            )
            val keyGenParameterSpec = KeyGenParameterSpec.Builder(
                keyAlias,
                KeyProperties.PURPOSE_ENCRYPT or KeyProperties.PURPOSE_DECRYPT
            )
                .setBlockModes(KeyProperties.BLOCK_MODE_GCM)
                .setEncryptionPaddings(KeyProperties.ENCRYPTION_PADDING_NONE)
                .setKeySize(256)
                .build()
            
            keyGenerator.init(keyGenParameterSpec)
            keyGenerator.generateKey()
        }
        
        val secretKeyEntry = keyStore.getEntry(keyAlias, null) as KeyStore.SecretKeyEntry
        return secretKeyEntry.secretKey
    }
    
    fun encrypt(data: String): String {
        val cipher = Cipher.getInstance("AES/GCM/NoPadding")
        cipher.init(Cipher.ENCRYPT_MODE, getSecretKey())
        
        val iv = cipher.iv
        val encrypted = cipher.doFinal(data.toByteArray())
        
        // 将IV和加密数据组合
        val combined = ByteArray(iv.size + encrypted.size)
        System.arraycopy(iv, 0, combined, 0, iv.size)
        System.arraycopy(encrypted, 0, combined, iv.size, encrypted.size)
        
        return Base64.encodeToString(combined, Base64.NO_WRAP)
    }
    
    fun decrypt(encryptedData: String): String {
        val combined = Base64.decode(encryptedData, Base64.NO_WRAP)
        
        val cipher = Cipher.getInstance("AES/GCM/NoPadding")
        val iv = ByteArray(12)  // GCM IV长度
        val encrypted = ByteArray(combined.size - iv.size)
        
        System.arraycopy(combined, 0, iv, 0, iv.size)
        System.arraycopy(combined, iv.size, encrypted, 0, encrypted.size)
        
        val spec = GCMParameterSpec(128, iv)
        cipher.init(Cipher.DECRYPT_MODE, getSecretKey(), spec)
        
        return String(cipher.doFinal(encrypted))
    }
}
```

```kotlin
private fun startServiceAndFinish() {
    Log.d(TAG, "✅ 所有权限已具备，检查激活状态...")
    
    lifecycleScope.launch {
        try {
            val activationManager = LicenseActivationManager.getInstance()

                // 验证成功，启动服务
                Log.d(TAG, "✅ 激活验证成功，启动服务")
                doStartService()
            } else if (result.needLicense) {
                // 需要激活码，显示对话框

        } catch (e: Exception) {
            Log.e(TAG, "❌ 激活检查失败: ${e.message}", e)
            // 🔒 出错时也阻止启动
            showActivationRequiredDialog()
        }
    }
}

        .setPositiveButton("重试") { _, _ ->
            startServiceAndFinish()
        }
        .setNegativeButton("退出") { _, _ ->
            finish()
        }
        .setCancelable(false)
        .show()
}
```

### 4. 实现定期验证机制

#### 4.1 添加定期验证方法

```kotlin
// LicenseActivationManager.kt
private var lastVerificationTime: Long = 0
private val VERIFICATION_INTERVAL = 24 * 60 * 60 * 1000L  // 24小时

suspend fun verifyPeriodically(): ActivationResult {
    val now = System.currentTimeMillis()
    val lastVerified = prefs.getLong("last_verification_time", 0)
    
    // 如果距离上次验证不足24小时，跳过
    if (now - lastVerified < VERIFICATION_INTERVAL) {
        Log.d(TAG, "距离上次验证不足24小时，跳过定期验证")
        return ActivationResult(activated = true, fromCache = true)
    }
    
    Log.d(TAG, "开始定期验证...")
    val result = checkOnStartup()
    
    if (result.activated) {
        prefs.edit().putLong("last_verification_time", now).apply()
    }
    
    return result
}
```

#### 4.2 在服务中定期验证

```kotlin
// EnhancedFloatingWindowService.kt
private val verificationHandler = Handler(Looper.getMainLooper())
private val verificationRunnable = object : Runnable {
    override fun run() {
        lifecycleScope.launch {
            val activationManager = LicenseActivationManager.getInstance()
            val result = activationManager.verifyPeriodically()
            
            if (!result.activated) {
                // 验证失败，停止服务
                Log.e(TAG, "定期验证失败，停止服务")
                stopSelf()
                // 通知用户需要重新激活
                showActivationNotification()
            }
            
            // 24小时后再次验证
            verificationHandler.postDelayed(this, 24 * 60 * 60 * 1000L)
        }
    }
}

override fun onCreate() {
    super.onCreate()
    // 启动定期验证
    verificationHandler.postDelayed(verificationRunnable, 24 * 60 * 60 * 1000L)
}
```

```kotlin
data class DeviceFingerprint(
    val mac: String?,
    val androidId: String,
    val serialNumber: String?,

    val deviceManufacturer: String

fun getDeviceFingerprint(): DeviceFingerprint {
    val mac = getMacAddress()
    val androidId = android.provider.Settings.Secure.getString(
        context.contentResolver,
        android.provider.Settings.Secure.ANDROID_ID
    ) ?: "UNKNOWN"
    
    val serialNumber = if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) {
        Build.getSerial()
    } else {
        @Suppress("DEPRECATION")
        Build.SERIAL
    }
    
    return DeviceFingerprint(
        mac = mac,
        androidId = androidId,
        serialNumber = serialNumber,
        buildFingerprint = Build.FINGERPRINT,
        deviceModel = Build.MODEL,
        deviceManufacturer = Build.MANUFACTURER
    )
}

fun getDeviceFingerprintHash(): String {
    val fingerprint = getDeviceFingerprint()
    val combined = "${fingerprint.mac}|${fingerprint.androidId}|${fingerprint.serialNumber}|${fingerprint.buildFingerprint}"
    return combined.sha256()  // 需要实现SHA256哈希
}
```
